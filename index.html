const SHEET_NAME = 'Registros'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (data._action === 'edit' && data._rowIndex) { const row = parseInt(data._rowIndex) + 1; sh.getRange(row, 10).setValue(data.entrada); sh.getRange(row, 11).setValue(data.salida); sh.getRange(row, 12).setValue(data.alm_ini); sh.getRange(row, 13).setValue(data.alm_fin); sh.getRange(row, 14).setValue(data.horas_efectivas); sh.getRange(row, 15).setValue(data.observaciones); sh.getRange(row, 17).setValue('Editado ' + new Date().toLocaleString()); return resp({ok:true}); } if (data._action === 'delete' && data._rowIndex) { const row2 = parseInt(data._rowIndex) + 1; sh.deleteRow(row2); return resp({ok:true}); } sh.appendRow([new Date(),data.fecha,data.proyecto,data.supervisor,data.sector, data.lat,data.lng,data.trabajador,data.cargo,data.entrada,data.salida, data.alm_ini,data.alm_fin,data.horas_efectivas,data.observaciones,data.tipo,'']); return resp({ok:true}); } catch(err) { return resp({ok:false,error:err.message}); } } function doGet(e) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); return resp(sh.getDataRange().getValues()); } function resp(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
